In Listener 2 the Listener's work is declared in a Process root.


Settle and Deliver are smaller pieces of work, and Deliver names itself: it goes on for as long as a text is deliverable. Where an input and its output bear one name, Jobs, History, Configuration, the head is the input and the last position the output.

```rust
impl Operable for Hear {
    type Outcome = Hearing;
    async fn operate(self, scope: &Scope) -> Hearing {
        let Hear(job_id) = self;
        let trial = scope.run(Tries(job_id.clone())).await;
        let mut tries = Vec::new();
        for backend in trial.backend_vector {
            let backend_name = backend.backend_name.clone();
            let transcribe = Transcribe { job_id: job_id.clone(), backend, term_vector: trial.term_vector.clone() };
            let failing = match scope.send(transcribe).await {
                Ok(Transcription::Heard(heard)) => {
                    let transcript = Transcript { backend_name, provider_id_option: heard.provider_id_option, text: heard.text };
                    return Hearing::Heard(Heard_Data { transcript, try_vector: tries });
                }
                Ok(Transcription::Failure(failure)) => Failing::Failure(failure),
                Err(unobserved) => Failing::Unobserved(unobserved),
            };
            let failed = Try { backend_name, failing };
            scope.send(Notify(Notice::BackendFailed(failed.clone()))).await.ok();
            tries.push(failed);
        }
        scope.send(Notify(Notice::Exhausted(job_id))).await.ok();
        Hearing::Exhausted(tries)
    }
}
```
